```python
#!/usr/bin/env python3
"""F3a 只读实证（task_api_embedder_internal_batching TC-EB-5）：
从本地 PG ops_knowledge_chunks 只读取样 >=1000 条真实 content，单次 ApiEmbedder().embed()
调用链实证不超时；断言 返回数==输入数 · 维数全 1024 · 批数台账 == ceil(N/128) · 保序抽对。
一次性片段（不落 scripts/ · 零写面：PG 仅 SELECT）。
"""
from __future__ import annotations

import logging
import math
import os
import sys
import time

sys.path.insert(0, os.getcwd())

from dotenv import load_dotenv

load_dotenv("/Users/cyning/Desktop/Ops-desk/ops-desk-api-worktrees/t-embedder-batch/.env")

logging.basicConfig(level=logging.INFO, format="%(levelname)s %(name)s %(message)s")
logging.getLogger("app.ops.rag.embedder").setLevel(logging.INFO)

from sqlalchemy import create_engine, text  # noqa: E402

from app.ops.rag.embedder import ApiEmbedder  # noqa: E402

N = 1200  # >=1000 裁定下限 · ceil(1200/128)=10 批


def main() -> int:
    url = os.environ["DATABASE_URL"].replace("+asyncpg", "+psycopg")
    engine = create_engine(url)
    with engine.connect() as conn:
        rows = conn.execute(
            text("SELECT content FROM ops_knowledge_chunks ORDER BY id LIMIT :n"),
            {"n": N},
        ).fetchall()
    texts = [r[0] for r in rows]
    assert len(texts) == N, f"取样不足: {len(texts)}"
    print(f"[F3a] sampled {len(texts)} chunks from PG (read-only)")

    emb = ApiEmbedder()
    print(f"[F3a] batch_size={emb._batch_size} (default expected 128)")
    t0 = time.monotonic()
    vectors = emb.embed(texts)  # 单次调用链 · 内部批切
    elapsed = time.monotonic() - t0

    assert len(vectors) == len(texts), f"返回数 {len(vectors)} != 输入数 {len(texts)}"
    dims = {len(v) for v in vectors}
    assert dims == {1024}, f"维数异常: {dims}"
    batches = math.ceil(len(texts) / emb._batch_size)
    print(
        f"[F3a] embed ok: {len(vectors)} vectors dim=1024 · expected batches={batches} "
        f"· total {elapsed:.1f}s · {len(texts)/elapsed:.1f} chunks/s"
    )

    # 保序抽对：跨批边界两侧 + 首尾 + 中段，单条重嵌做余弦比对
    # （API 批内数值有非确定性：逐分量差可至 ~1e-3，错位则余弦 <<0.9；
    #  判定 = 同文本余弦 >0.999 且显著高于邻位文本余弦）
    def cos(a, b):
        dot = sum(x * y for x, y in zip(a, b, strict=True))
        na = math.sqrt(sum(x * x for x in a))
        nb = math.sqrt(sum(x * x for x in b))
        return dot / (na * nb)

    probes = sorted({0, 127, 128, len(texts) // 2, len(texts) - 1})
    for i in probes:
        single = emb.embed([texts[i]])[0]
        same = cos(single, vectors[i])
        neigh = max(cos(vectors[i], vectors[j]) for j in (i - 1, i + 1) if 0 <= j < len(vectors))
        print(f"[F3a] order probe idx={i}: cos(same)={same:.6f} cos(neighbor)={neigh:.6f}")
        assert same > 0.999 and same > neigh + 0.01, f"保序违例 idx={i} same={same} neigh={neigh}"
    print("[F3a] order probes all pass (cosine criterion) · verdict=PASS")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
